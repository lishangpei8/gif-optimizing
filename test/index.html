📖 使用指南

如何使用:

  1. 使用转换脚本将GIF转换为WebM和HEVC格式
  2. 将转换后的文件放入 test/assets/ 目录
  3. 在不同设备/浏览器上打开此页面测试
  4. 根据测试结果选择最佳格式
<!-- H5页面中的推荐实现 -->
<video autoplay loop muted playsinline>
  <!-- iOS优先使用HEVC -->
  <source src="animation.mp4" type='video/mp4; codecs="hvc1"'>
  <!-- Android/Desktop使用WebM -->
  <source src="animation.webm" type="video/webm">
  <!-- Fallback使用GIF -->
  <img src="animation.gif" alt="Animation">
</video>
// JavaScript动态选择最佳格式
const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent);
const isAndroid = /Android/.test(navigator.userAgent);

let videoSrc;
if (isIOS) {
  videoSrc = 'animation.mp4';  // HEVC for iOS
} else if (isAndroid) {
  videoSrc = 'animation.webm'; // WebM for Android
} else {
  videoSrc = 'animation.webm'; // WebM for Desktop
}
